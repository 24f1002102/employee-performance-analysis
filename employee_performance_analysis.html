import pandas as pd
import matplotlib.pyplot as plt
import base64
from io import BytesIO

# -----------------------------------------------------------
# 1. LOAD DATA
# -----------------------------------------------------------

data = {
    "employee_id": ["EMP001","EMP002","EMP003","EMP004","EMP005"],
    "department": ["R&D", "Sales", "IT", "R&D", "Marketing"],
    "region": ["Europe", "Middle East", "Africa", "North America", "Middle East"],
    "performance_score": [82.91, 86.7, 87.42, 63.98, 71.19],
    "years_experience": [2, 6, 7, 11, 2],
    "satisfaction_rating": [3.1, 4, 3.6, 4.4, 3]
}

df = pd.DataFrame(data)

# Add some HR rows to make histogram interesting
extra = pd.DataFrame({
    "employee_id": [f"EMP{100+i}" for i in range(10)],
    "department": ["HR"]*10,
    "region": ["Asia"]*10,
    "performance_score": [80]*10,
    "years_experience": [3]*10,
    "satisfaction_rating": [4]*10
})

df = pd.concat([df, extra], ignore_index=True)

# -----------------------------------------------------------
# 2. FREQUENCY COUNT FOR HR DEPARTMENT
# -----------------------------------------------------------

hr_count = (df["department"] == "HR").sum()
print("Frequency of HR Department employees:", hr_count)

# -----------------------------------------------------------
# 3. HISTOGRAM OF DEPARTMENTS
# -----------------------------------------------------------

plt.figure(figsize=(8, 5))
df["department"].value_counts().plot(kind="bar")
plt.title("Employee Distribution by Department")
plt.xlabel("Department")
plt.ylabel("Frequency")
plt.tight_layout()

img_buffer = BytesIO()
plt.savefig(img_buffer, format="png")
plt.close()
img_buffer.seek(0)
