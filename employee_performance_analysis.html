import pandas as pd
import matplotlib.pyplot as plt
import random
import base64
from io import BytesIO

# -----------------------------------------------------------
# 1. CREATE A SAMPLE DATASET (100 employees)
# -----------------------------------------------------------

departments = ["HR", "Sales", "Marketing", "Finance", "IT", "Operations"]
regions = ["North", "South", "East", "West"]

data = {
    "Employee_ID": range(1, 101),
    "Department": [random.choice(departments) for _ in range(100)],
    "Region": [random.choice(regions) for _ in range(100)],
    "Performance_Score": [random.randint(1, 10) for _ in range(100)]
}

df = pd.DataFrame(data)

# -----------------------------------------------------------
# 2. CALCULATE FREQUENCY COUNT FOR HR DEPARTMENT
# -----------------------------------------------------------

hr_count = (df["Department"] == "HR").sum()
print("Frequency of HR Department employees:", hr_count)

# -----------------------------------------------------------
# 3. CREATE HISTOGRAM OF DEPARTMENTS
# -----------------------------------------------------------

plt.figure(figsize=(8, 5))
df["Department"].value_counts().plot(kind="bar")
plt.title("Employee Distribution by Department")
plt.xlabel("Department")
plt.ylabel("Frequency")
plt.tight_layout()

img_buffer = BytesIO()
plt.savefig(img_buffer, format="png")
plt.close()
img_buffer.seek(0)
